function publicUrl(value) {
  const url = new URL(String(value));
  if (!/^https?:$/.test(url.protocol) || url.username || url.password || !url.hostname.includes(".") || /^\d+\.\d+\.\d+\.\d+$/.test(url.hostname) || /[\[\]:]/.test(url.hostname) || /(?:^|\.)(?:localhost|local|internal|invalid|test|home|lan)$/.test(url.hostname) || /(?:^|\.)(?:nip\.io|sslip\.io|localtest\.me)$/.test(url.hostname)) {
    throw new Error("URL p\xFAblica http(s) requerida");
  }
  return url;
}
async function readBytes(response, account) {
  if (Number(response.headers.get("content-length")) > MAX_BODY_BYTES) throw new Error("Respuesta Seanime demasiado grande");
  if (!response.body) return new Uint8Array();
  const reader = response.body.getReader();
  const chunks = [];
  let length = 0;
  try {
    while (true) {
      const chunk = await reader.read();
      if (chunk.done) break;
      length += chunk.value.byteLength;
      if (length > MAX_BODY_BYTES) throw new Error("Respuesta Seanime demasiado grande");
      account(chunk.value.byteLength);
      chunks.push(chunk.value);
    }
  } finally {
    await reader.cancel().catch(() => {
    });
    reader.releaseLock();
  }
  const bytes = new Uint8Array(length);
  let offset = 0;
  for (const chunk of chunks) {
    bytes.set(chunk, offset);
    offset += chunk.length;
  }
  return bytes;
}
function cookieValues(headers) {
  const compatible = headers;
  if (compatible.getSetCookie) return compatible.getSetCookie();
  if (compatible.getAll) return compatible.getAll("Set-Cookie");
  return (headers.get("set-cookie") ?? "").split(/,\s*(?=[^=;,\s]+=)/).filter(Boolean);
}

var HARD_SUB = /\b(?:hard[ ._-]*sub(?:bed|s)?|burn(?:ed|t)[ ._-]*in|hardsubbed)\b/i;
var FOREIGN_DUB = /\b(?:dub(?:bed|s)?|dual[ ._-]*audio|multi[ ._-]*audio|english[ ._-]*audio|german[ ._-]*audio|french[ ._-]*audio)\b/i;
var RAW_OR_SOFT = /\b(?:raws?|soft[ ._-]*sub(?:bed|s)?|unsubbed|no[ ._-]*subs?)\b/i;
var KNOWN_SOFT_RELEASE = /^\s*\[(?:SubsPlease|Erai-raws|Ohys-Raws|Leopard-Raws)\]/i;
function allowsAdditionalTorrent(torrent, rawCategory = false) {
  const name = String(torrent.name ?? "");
  if (HARD_SUB.test(name) || torrent.hardsub === true || FOREIGN_DUB.test(name)) return false;
  const audio = String(torrent.audio_lang ?? torrent.audioLanguage ?? "").toLowerCase();
  if (audio && !/^(ja|jpn|japanese|es(?:-[a-z0-9]+)?|spa|spanish)$/.test(audio)) return false;
  return rawCategory || torrent.hardsub === false && Boolean(audio) || RAW_OR_SOFT.test(name) || KNOWN_SOFT_RELEASE.test(name);
}
function allowsAdditionalVideo(source, server, trustedSoft = false) {
  const label = [server, source.label, source.quality, source.subType].filter(Boolean).join(" ");
  if (HARD_SUB.test(label) || /\b(?:dub|english|hindi|tamil|telugu|german|french|portuguese|italian|korean|russian|arabic)\b/i.test(label)) return false;
  return /\b(?:soft(?:[ ._-]*sub)?|raw)\b/i.test(label) || trustedSoft && Array.isArray(source.subtitles) && source.subtitles.length > 0;
}

function normalizeTitle(value) {
  return value.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "").replace(/\b(?:audio\s+)?(?:latino|castellano|japones|subtitulado)\b/g, " ").replace(/[^a-z0-9]+/g, " ").trim();
}
function seasonNumber(title) {
  const numeric = title.match(/(\d+)\s*(?:st|nd|rd|th|ª)?\s*(?:season|temporada|cour)|\b(?:season|temporada|cour|s)\s*(\d+)/i);
  if (numeric) return Number(numeric[1] ?? numeric[2]);
  const roman = title.match(/(?:^|\s)(iv|v|iii|ii|i)(?:\s*:|$)/i);
  if (roman) return { i: 1, ii: 2, iii: 3, iv: 4, v: 5 }[roman[1].toLowerCase()];
  const trailing = title.match(/\b(\d{1,2})\s*$/);
  return trailing && Number(trailing[1]) <= 50 && !/\b(?:part|parte|movie|film)\s+\d+\s*$/i.test(title) ? Number(trailing[1]) : null;
}
function rankSearchResults(results, title, season, mediaType) {
  const wanted = normalizeTitle(title);
  const movie = /\b(movie|film|pelicula|ova|special|especial)\b/i;
  return (Array.isArray(results) ? results : []).filter((result) => result && typeof result.id === "string").map((result, index) => {
    const name = String(result.title ?? result.Title ?? "");
    const normalized = normalizeTitle(name);
    const parsedSeason = seasonNumber(name);
    if (mediaType !== "movie" && parsedSeason !== null && parsedSeason !== season) return null;
    let score = Math.max(0, 100 - index);
    if (normalized === wanted) score += 1e3;
    else if (normalized.startsWith(wanted)) score += 500;
    else if (normalized.includes(wanted) || wanted.includes(normalized) && normalized) score += 300;
    if (mediaType !== "movie" && movie.test(name) && !movie.test(title)) score -= 2e3;
    if (mediaType !== "movie" && parsedSeason === season) score += 2e3;
    if (season === 1 && /(?:-hen|final season)\s*$/i.test(name)) score -= 2e3;
    score -= normalized.length;
    return { result: { ...result, title: name }, score };
  }).filter((entry) => entry !== null).sort((a, b) => b.score - a.score).map((entry) => entry.result);
}
async function pool(items, concurrency, task) {
  const results = [];
  let next = 0;
  await Promise.all(Array.from({ length: Math.min(concurrency, items.length) }, async () => {
    while (next < items.length) {
      const index = next++;
      try {
        results[index] = await task(items[index]);
      } catch {
      }
    }
  }));
  return results.filter((value) => value !== void 0);
}
function audioMode(result, episode, requestedDub) {
  try {
    const parsed = JSON.parse(episode.id);
    if (parsed.type === "sub" || parsed.type === "dub") return parsed.type;
  } catch {
  }
  return result.subOrDub === "dub" || result.subOrDub === "both" && requestedDub ? "dub" : "sub";
}
function playbackHeaders(meta, headers) {
  const merged = Object.fromEntries(Object.entries(headers).filter(([, value]) => typeof value === "string"));
  if (meta.id === "animeav1") {

      merged["Sec-Fetch-Dest"] = "empty";

}
async function online(provider, meta, media, title, season, number, type) {

    if (!result || !provider.findEpisodes || !provider.findEpisodeServer) return [];
    const episodes = await provider.findEpisodes(result.id);
    const episode = episodes.find((item) => Number(item.number) === number);
    if (!episode) return [];

    const found = await pool(selected, 3, (server) => provider.findEpisodeServer(episode, server));

      if (typeof source.url !== "string" || !["mp4", "m3u8", "hls", "mpd", "mkv"].includes(String(source.type).toLowerCase())) return [];

      const headers = playbackHeaders(meta, { ...server.headers, ...source.headers });
      return [{
        name: meta.name,
        sourceName: meta.name,
        title: [server.server, source.quality, mode === "dub" ? "Audio Espa\xF1ol" : strict ? "Audio Japon\xE9s \xB7 subt\xEDtulos opcionales" : "Sub: Espa\xF1ol"].filter(Boolean).join(" \xB7 "),

function torrentHash(value) {
  const input = String(value ?? "").trim();
  if (/^[a-f\d]{40}$/i.test(input)) return input.toLowerCase();
  const hash = input.match(/[?&]xt=urn(?::|%3A)btih(?::|%3A)([a-z0-9]{32,40})(?:&|$)/i)?.[1] ?? input;
  if (/^[a-f\d]{40}$/i.test(hash)) return hash.toLowerCase();
  if (!/^[a-z2-7]{32}$/i.test(hash)) return null;
  let bits = 0, accumulator = 0, hex = "";
  for (const character of hash.toUpperCase()) {
    accumulator = accumulator << 5 | "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567".indexOf(character);
    bits += 5;
    if (bits >= 8) {
      bits -= 8;
      hex += (accumulator >>> bits & 255).toString(16).padStart(2, "0");
    }
  }
  return hex;
}
async function torrents(provider, meta, media, title, episode, type) {
  const settings = await provider.getSettings();
  const options = { media, query: title, batch: false, episodeNumber: episode, resolution: "", anidbAID: 0, anidbEID: 0, bestReleases: true };
  const raw = settings.canSmartSearch && provider.smartSearch ? await provider.smartSearch(options) : await provider.search(options);
  const candidates = (Array.isArray(raw) ? raw : []).filter((torrent) => {
    if (!torrent || !allowsAdditionalTorrent(torrent, meta.rawCategory)) return false;
    if (type === "movie") return true;
    const name = String(torrent.name ?? "");
    const releaseSeason = seasonNumber(name);
    if (releaseSeason !== null && releaseSeason !== Number(media.season ?? 1)) return false;
    const parsed = name.match(/\bS\d{1,2}E(\d{1,4})\b/i)?.[1] ?? name.match(/(?:\s-\s|\bEP?\s*)(\d{1,4})\b/i)?.[1];
    const number = Number(torrent.episodeNumber) > 0 ? Number(torrent.episodeNumber) : Number(parsed);
    return !torrent.isBatch && number === episode;
  }).slice(0, 12);
  return (await pool(candidates, 3, async (torrent) => {
    let magnet = String(torrent.magnetLink ?? "");
    let infoHash = torrentHash(torrent.infoHash) ?? torrentHash(magnet);
    if (!infoHash && provider.getTorrentInfoHash) infoHash = torrentHash(await provider.getTorrentInfoHash(torrent));
    if (!infoHash && provider.getTorrentMagnetLink) {
      magnet = await provider.getTorrentMagnetLink(torrent);
      infoHash = torrentHash(magnet);
    }
    if (!infoHash) return null;
    return {
      name: meta.name,
      title: String(torrent.name),
      infoHash,
      seeders: Number.isFinite(Number(torrent.seeders)) && Number(torrent.seeders) >= 0 ? Number(torrent.seeders) : void 0,
      size: Number(torrent.size) > 0 ? Number(torrent.size) : void 0,
      languages: ["ja"],
      sources: torrentHash(magnet) ? [magnet] : void 0,
      behaviorHints: { providerId: meta.id, providerName: meta.name, filename: String(torrent.name), subtitlePolicy: "raw-or-soft", mediaExtensionManifest: meta.manifestURI }
    };
  })).filter((stream) => stream !== null);
}
async function animeMedia(context, meta, title, type, season) {
  const media = { id: 0, romajiTitle: title, englishTitle: title, synonyms: [], isAdult: false, format: type === "movie" ? "MOVIE" : "TV", episodeCount: 0, absoluteSeasonOffset: 0, season };
  if (!meta.needsAnimeId) return media;
  const response = await context.fetch("https://graphql.anilist.co", { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify({
    query: "query($search:String){Page(perPage:8){media(search:$search,type:ANIME,isAdult:false){id idMal title{romaji english} synonyms format episodes startDate{year}}}}",
    variables: { search: title }
  }) });
  if (!response.ok) return media;
  const found = response.json()?.data?.Page?.media ?? [];
  const ranked = rankSearchResults(found.map((item) => ({ ...item, id: String(item.id), title: item.title?.romaji ?? item.title?.english })), title, season, type);
  const match = found.find((item) => String(item.id) === ranked[0]?.id);
  return match ? { ...media, id: match.id, idMal: match.idMal, romajiTitle: match.title.romaji || title, englishTitle: match.title.english || title, synonyms: match.synonyms ?? [], episodeCount: match.episodes ?? 0, format: match.format, startDate: match.startDate } : media;
}
function createSeanimeAdapter(factory, meta, dependencies = {}) {
  return {
    async getStreams(first, ...rest) {
      const args = Array.isArray(first) ? first : [first, ...rest];
      const [, inputType, rawSeason, rawEpisode, rawTitle] = args;
      const type = inputType === "series" || inputType === "anime" ? "tv" : inputType;
      const title = typeof rawTitle === "string" ? rawTitle.trim().slice(0, 300) : "";
      const season = Number(rawSeason ?? 1);
      const episode = type === "movie" ? 1 : Number(rawEpisode ?? 1);
      if (!title || !["movie", "tv"].includes(String(type)) || !Number.isInteger(episode) || episode < 1 || !Number.isInteger(season) || season < 1) return [];
      const controller = new AbortController();
      const timeout = setTimeout(() => controller.abort(), 12e3);
      try {
        const context = createContext(meta, dependencies, controller.signal);
        const provider = factory(context);
        const media = await animeMedia(context, meta, title, String(type), season);
        if (meta.needsAnimeId && !media.id) return [];
        const streams = meta.type === "anime-torrent-provider" ? await torrents(provider, meta, media, title, episode, String(type)) : await online(provider, meta, media, title, season, episode, String(type));
        const seen = /* @__PURE__ */ new Set();
        return streams.filter((stream) => {
          const key = String(stream.url ?? stream.infoHash);
          if (seen.has(key)) return false;
          seen.add(key);
          return true;
        }).slice(0, 12);
      } finally {
        controller.abort();
        clearTimeout(timeout);
      }
    }
  };
