/// <reference path="./anime-torrent-provider.d.ts" />

class Provider {

    constructor() {
        this._MAX_SEASONAL_ANIME = 15
        this._MAX_LATEST_RESULTS = 30
        this._MAX_SEARCH_LIMIT = 5
    }

    // ================================================================
    //  Required: Provider settings
    // ================================================================
    getSettings() {
        return {
            canSmartSearch: true,
            smartSearchFilters: ["batch", "episodeNumber", "resolution", "query", "bestReleases"],
            supportsAdult: false,
            type: "main",
        }
    }

    // ================================================================
    //  Required: Text search (from user query bar)
    // ================================================================
    async search(opts) {
        try {
            const sel = await this._selectBestBgmId(opts)
            if (!sel || !sel.items.length) return []
            return sel.items.map(r => this._toAnimeTorrent(r, opts.media)).filter(Boolean)
        } catch (e) {
            console.error("[AniBT] search error:", e)
            return []
        }
    }

    // ================================================================
    //  Required: Structured smart search (Auto Downloader)
    // ================================================================
    async smartSearch(opts) {
        try {
            const sel = await this._selectBestBgmId(opts)
            if (!sel || !sel.items.length) return []

            // Convert ALL releases FIRST: episode numbers and batch flags are
            // resolved from episodeKey + $habari title parsing, so no release
            // is dropped before the filters can evaluate it.
            let out = sel.items.map(r => this._toAnimeTorrent(r, opts.media)).filter(Boolean)

            // ── episodeNumber / batch ──
            if (opts.batch) {
                // batch mode: only multi-episode releases (BATCH marker / episode ranges)
                out = out.filter(t => t.isBatch)
            } else if (opts.episodeNumber > 0) {
                // single episode: keep releases that actually contain the episode.
                // AniBT keys are inconsistent for sequels — some groups use
                // relative numbers (1-10), others absolute (29-38) — match both
                out = out.filter(t => this._matchesEpisode(t, opts.episodeNumber, opts.media))
            }

            // ── resolution (normalized) ──
            // Fall back to all releases if none match, so results are never
            // zeroed out over a resolution string mismatch (e.g. "1080" vs "1080p")
            if (opts.resolution) {
                const target = opts.resolution.toLowerCase().replace(/p$/, "")
                const matched = out.filter(t =>
                    (t.resolution || "").toLowerCase().replace(/p$/, "") === target
                )
                if (matched.length) out = matched
            }

            // ── bestReleases: return ONLY the highest-quality torrent ──
            // (AniBT does not expose seeders, so sort by date as a tiebreaker)
            if (opts.bestReleases && out.length) {
                out.sort((a, b) =>
                    (b.seeders - a.seeders) ||
                    ((Date.parse(b.date || "") || 0) - (Date.parse(a.date || "") || 0))
                )
                const best = out[0]
                best.isBestRelease = true
                return [best]
            }

            return out
        } catch (e) {
            console.error("[AniBT] smartSearch error:", e)
            return []
        }
    }

    // ================================================================
    //  Required: Torrent hash / magnet resolution
    // ================================================================
    async getTorrentInfoHash(torrent) {
        if (torrent.infoHash) return torrent.infoHash
        if (torrent.magnetLink) {
            const m = torrent.magnetLink.match(/btih:([a-fA-F0-9]{40})/i)
            if (m) return m[1].toLowerCase()
        }
        return ""
    }

    async getTorrentMagnetLink(torrent) {
        return torrent.magnetLink || ""
    }

    // ================================================================
    //  Required: Latest torrents (home page / discover)
    // ================================================================
    async getLatest() {
        try {
            const r = await fetch("https://anibt.net/api/seasons/anime")
            if (!r || !r.ok) return []
            const data = await r.json()
            if (!data || !data.data || !data.data.byWeekday) return []

            const bgmIds = []
            for (let w = 0; w < data.data.byWeekday.length; w++) {
                const animes = data.data.byWeekday[w].animes || []
                for (let a = 0; a < animes.length; a++) {
                    const s = animes[a]
                    if (s.hasRelease && s.bgmId) {
                        bgmIds.push(s.bgmId)
                        if (bgmIds.length >= this._MAX_SEASONAL_ANIME) break
                    }
                }
                if (bgmIds.length >= this._MAX_SEASONAL_ANIME) break
            }

            // ── Concurrently fetch releases for all seasonal anime ──
            // _fetchReleases handles its own errors (returns []), so Promise.all is safe
            const all = []
            const results = await Promise.all(
                bgmIds.map(id =>
                    this._fetchReleases(id).catch(() => [])
                )
            )
            for (let i = 0; i < results.length; i++) {
                const items = results[i]
                for (let j = 0; j < items.length; j++) {
                    all.push(items[j])
                }
            }

            // Sort by newest first, limit to _MAX_LATEST_RESULTS
            all.sort((a, b) => (b.publishedAt || 0) - (a.publishedAt || 0))
            const limit = Math.min(all.length, this._MAX_LATEST_RESULTS)
            const latest = []
            for (let l = 0; l < limit; l++) {
                latest.push(this._toAnimeTorrent(all[l], null))
            }
            return latest
        } catch (e) {
            console.error("[AniBT] getLatest error:", e)
            return []
        }
    }

    // ================================================================
    //  Private: BGM ID resolution (deduplicated from search/smartSearch)
    // ================================================================

    /**
     * Resolve candidate BGM IDs, fetch their releases, and pick the best match.
     *
     * The search API ranks subjects by rating, so it can put the WRONG season
     * first (e.g. Frieren S1 — which has no releases — ahead of S2). Instead of
     * blindly using the first result, we compare release coverage and prefer a
     * candidate whose releases contain the requested episode.
     */
    async _selectBestBgmId(opts) {
        const ids = await this._resolveBgmIds(opts)
        if (!ids.length) return { bgmId: 0, items: [] }

        const target = opts && opts.episodeNumber > 0 ? opts.episodeNumber : 0
        const offset = (opts && opts.media && opts.media.absoluteSeasonOffset) || 0

        let best = null
        for (let i = 0; i < ids.length; i++) {
            const items = await this._fetchReleases(ids[i])

            let score = items.length
            if (target > 0) {
                for (let j = 0; j < items.length; j++) {
                    if (this._rawContainsEpisode(items[j], target, offset)) {
                        score += 1000 // an episode match dominates release count
                        break
                    }
                }
            }

            if (!best || score > best.score) {
                best = { bgmId: ids[i], items: items, score: score }
            }

            // early exit: a candidate already covers the requested episode
            if (best.score >= 1000) break
        }

        return best ? { bgmId: best.bgmId, items: best.items } : { bgmId: 0, items: [] }
    }

    /** Build query candidates and return the matching Bangumi IDs (deduped) */
    async _resolveBgmIds(opts) {
        const queries = this._buildQueries(opts)
        if (!queries.length) return []

        // 如果搜索词是纯数字，直接作为 Bangumi ID 使用，跳过 API 搜索
        for (const q of queries) {
            const trimmed = q.trim()
            if (/^\d+$/.test(trimmed)) {
                const id = parseInt(trimmed, 10)
                if (id > 0) {
                    console.log("[AniBT] 直接使用 Bangumi ID:", id)
                    return [id]
                }
            }
        }

        const ids = []
        const seen = {}
        for (const q of queries) {
            const arr = await this._searchBgmIds(q)
            for (let i = 0; i < arr.length && ids.length < this._MAX_SEARCH_LIMIT; i++) {
                if (!seen[arr[i]]) {
                    seen[arr[i]] = true
                    ids.push(arr[i])
                }
            }
            if (ids.length >= this._MAX_SEARCH_LIMIT) break
        }
        return ids
    }

    // ================================================================
    //  Private: API helpers
    // ================================================================

    /** Search for Bangumi IDs by anime title via AniBT search API (all results) */
    async _searchBgmIds(query) {
        if (!query) return []
        const url = "https://anibt.net/api/bgm/search?q=" + encodeURIComponent(query) +
                     "&limit=" + this._MAX_SEARCH_LIMIT
        try {
            const r = await fetch(url)
            if (!r || !r.ok) return []
            const d = await r.json()
            const arr = d && d.data || d
            const ids = []
            if (arr && arr.length) {
                for (let i = 0; i < arr.length; i++) {
                    if (arr[i] && arr[i].bgmId) ids.push(arr[i].bgmId)
                }
            }
            return ids
        } catch (e) {
            console.error("[AniBT] 搜索失败:", query, e)
            return []
        }
    }

    /** Fetch all release items for a Bangumi ID from AniBT groups API */
    async _fetchReleases(bgmId) {
        if (!bgmId) return []
        const url = "https://anibt.net/api/anime/groups?bgmId=" + bgmId
        try {
            const r = await fetch(url)
            if (!r || !r.ok) return []
            const d = await r.json()
            if (!d || !d.ok || !d.data) return []

            const out = []
            const groups = d.data.groups || []
            for (let g = 0; g < groups.length; g++) {
                const group = groups[g]
                const items = group.items || []
                for (let i = 0; i < items.length; i++) {
                    const it = items[i]
                    it._groupName = group.name || ""
                    out.push(it)
                }
            }
            return out
        } catch (e) {
            console.error("[AniBT] 获取发布失败:", bgmId, e)
            return []
        }
    }

    // ================================================================
    //  Private: Query building
    // ================================================================

    /** Build search query candidates from search/smartSearch options */
    _buildQueries(opts) {
        const m = opts && opts.media
        if (!m) {
            const q = opts && opts.query
            return q ? [q.trim()] : []
        }

        const seen = {}
        const q = []
        const add = s => {
            if (s && !seen[s]) { seen[s] = true; q.push(s) }
        }

        // Priority: user query > romaji > english > synonyms
        if (opts.query) add(opts.query.trim())
        if (m.romajiTitle) add(m.romajiTitle)
        if (m.englishTitle) add(m.englishTitle)
        if (m.synonyms) {
            for (let i = 0; i < m.synonyms.length && q.length < 5; i++) {
                add(m.synonyms[i])
            }
        }

        return q
    }

    // ================================================================
    //  Private: Data transformation
    // ================================================================

    /**
     * Convert a raw AniBT API item to AnimeTorrent format.
     *
     * episodeKey formats seen in the wild:
     *   "5"        → single episode
     *   "1-10"     → multi-episode range (batch)
     *   "BATCH"    → batch marker (range usually parseable from the title)
     *   "MOVIE"    → movie (resolved to episode 1 for movie media)
     *   missing    → unknown, resolved via $habari title parsing
     */
    _toAnimeTorrent(r, media) {
        if (!r) return null

        // Parse title with $habari for episode/resolution/group extraction
        let meta = null
        try { meta = $habari.parse(r.title) } catch (_) { /* ignore */ }

        // ── Episode number, range & batch detection ──
        let ep = -1
        let range = null
        let isBatch = false
        const key = r.episodeKey != null ? String(r.episodeKey).trim() : ""
        const keyUpper = key.toUpperCase()

        if (/^\d+$/.test(key)) {
            ep = parseInt(key, 10)
            if (ep === 0) { ep = -1; isBatch = true } // "0" = batch marker
        } else if (/^\d+\s*-\s*\d+$/.test(key)) {
            const [lo, hi] = key.split("-").map(s => parseInt(s.trim(), 10))
            ep = lo
            range = [lo, hi]
            isBatch = true
        } else if (keyUpper === "BATCH") {
            isBatch = true
        } else if (keyUpper === "MOVIE") {
            // single movie — resolved to episode 1 below when media is a movie
        } else if (!key) {
            isBatch = true // unknown span — confirmed below via title parsing
        } else {
            isBatch = true // unparseable key (specials, etc.)
        }

        // $habari title fallback: single episode or range (e.g. "[01-12 Fin]")
        if (meta && meta.episode_number != null) {
            const nums = []
            const arr = Array.isArray(meta.episode_number) ? meta.episode_number : [meta.episode_number]
            for (let i = 0; i < arr.length; i++) {
                const n = parseInt(String(arr[i]), 10)
                if (!isNaN(n)) nums.push(n)
            }
            if (nums.length >= 2) {
                range = [Math.min(...nums), Math.max(...nums)]
                isBatch = true
                if (ep === -1) ep = range[0]
            } else if (nums.length === 1) {
                if (ep === -1) {
                    ep = nums[0]
                    // missing key + single episode in the title → single release
                    if (!key) isBatch = false
                }
            }
        }

        // Info hash from magnet link
        let hash = ""
        if (r.magnet) {
            const mh = r.magnet.match(/btih:([a-fA-F0-9]{40})/i)
            if (mh) hash = mh[1].toLowerCase()
        }

        // ISO date string
        let date = ""
        if (r.publishedAt) {
            try { date = new Date(r.publishedAt).toISOString() } catch (_) { /* ignore */ }
        }

        // Resolution: prefer API field, fall back to $habari (video_resolution)
        let resolution = r.resolution || ""
        if (!resolution && meta && meta.video_resolution) {
            resolution = meta.video_resolution
        }

        // Release group: prefer API group name, fall back to $habari (release_group)
        let releaseGroup = r._groupName || ""
        if (!releaseGroup && meta && meta.release_group) {
            releaseGroup = meta.release_group
        }

        // Movie/single-episode special case: if no episode detected and media is MOVIE
        // or single-episode, default to episode 1 (per official reference implementation)
        if (!isBatch && ep === -1 && media && (media.episodeCount === 1 || media.format === "MOVIE")) {
            ep = 1
        }

        return {
            name: r.title || "",
            date: date,
            size: r.size || 0,
            formattedSize: "",
            seeders: r.seeders != null ? r.seeders : -1,
            leechers: r.leechers != null ? r.leechers : -1,
            downloadCount: r.downloads || 0,
            link: r.link || r.url || "",
            downloadUrl: r.downloadUrl || "",
            magnetLink: r.magnet || "",
            infoHash: hash,
            resolution: resolution,
            isBatch: isBatch,
            episodeNumber: ep,
            episodeRange: range, // internal: [min, max] for multi-episode releases
            releaseGroup: releaseGroup,
            isBestRelease: false,
            confirmed: true,
        }
    }

    // ================================================================
    //  Private: Episode matching
    // ================================================================

    /**
     * Whether a converted release contains the requested (relative) episode.
     *
     * AniBT follows bgm.tv numbering for sequels, which is inconsistent:
     * some groups number relative to the season (1-10), others continue from
     * the previous season (29-38). Seanime passes media.absoluteSeasonOffset
     * (e.g. 28 for Frieren S2), so we match both conventions — this mirrors
     * Seanime's own isProbablySameEpisode check.
     */
    _matchesEpisode(t, episodeNumber, media) {
        if (!t || episodeNumber <= 0) return false

        const offset = (media && media.absoluteSeasonOffset) || 0
        const targets = [episodeNumber]
        if (offset > 0 && targets.indexOf(episodeNumber + offset) < 0) {
            targets.push(episodeNumber + offset)
        }

        // exact episode number (from episodeKey or title)
        if (t.episodeNumber > 0 && targets.indexOf(t.episodeNumber) >= 0) return true

        // multi-episode range (e.g. "1-10", "25-35")
        if (t.episodeRange) {
            const lo = t.episodeRange[0]
            const hi = t.episodeRange[1]
            for (let i = 0; i < targets.length; i++) {
                if (targets[i] >= lo && targets[i] <= hi) return true
            }
        }

        return false
    }

    /** Raw-item variant of _matchesEpisode, used for candidate scoring (episodeKey only) */
    _rawContainsEpisode(r, episodeNumber, offset) {
        if (!r || episodeNumber <= 0) return false

        const targets = [episodeNumber]
        if (offset > 0 && targets.indexOf(episodeNumber + offset) < 0) {
            targets.push(episodeNumber + offset)
        }

        const key = r.episodeKey != null ? String(r.episodeKey).trim() : ""
        if (/^\d+$/.test(key)) {
            const n = parseInt(key, 10)
            if (targets.indexOf(n) >= 0) return true
        } else if (/^\d+\s*-\s*\d+$/.test(key)) {
            const [lo, hi] = key.split("-").map(s => parseInt(s.trim(), 10))
            for (let i = 0; i < targets.length; i++) {
                if (targets[i] >= lo && targets[i] <= hi) return true
            }
        }
        return false
    }

    // ================================================================
    //  Private: Utilities
    // ================================================================

}
